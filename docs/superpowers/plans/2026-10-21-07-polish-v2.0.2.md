```python
"""Garantiza una sola instancia POR USUARIO. Si ya hay una abierta, le pide que se muestre.

Usa QLocalServer/QLocalSocket (named pipe en Windows): la 1ª instancia abre un
servidor; las siguientes detectan que ya existe, le envían un mensaje para que
traiga su ventana al frente, y se cierran.

La clave incluye el usuario: en Windows el named pipe es global a la máquina, y
con varias sesiones (RDP / cambio rápido de usuario) el usuario B le mandaría
"show" a la instancia del usuario A y se cerraría sin abrir la suya.
"""
from __future__ import annotations

import getpass
import re
from typing import Callable

from PySide6.QtNetwork import QLocalServer, QLocalSocket

_KEY_PREFIX = "NetGauge.single-instance.v1"
_PING = b"show"
_CONNECT_TIMEOUT_MS = 1000


def build_key(user: str | None = None) -> str:
    """Clave del pipe: prefijo fijo + usuario sanitizado (solo [A-Za-z0-9_.-])."""
    if user is None:
        try:
            user = getpass.getuser()
        except Exception:  # noqa: BLE001 — sin usuario resoluble: clave genérica
            user = "user"
    safe = re.sub(r"[^A-Za-z0-9_.-]", "_", user or "") or "user"
    return f"{_KEY_PREFIX}.{safe}"


class SingleInstance:
    def __init__(self, key: str | None = None,
                 server_factory: Callable[[], QLocalServer] = QLocalServer) -> None:
        self._key = key or build_key()
        self._server_factory = server_factory
        self._server: QLocalServer | None = None

    def already_running(self) -> bool:
        """True si otra instancia respondió. Le manda 'mostrate' de paso."""
        sock = QLocalSocket()
        sock.connectToServer(self._key)
        if sock.waitForConnected(_CONNECT_TIMEOUT_MS):
            sock.write(_PING)
            sock.flush()
            sock.waitForBytesWritten(_CONNECT_TIMEOUT_MS)
            sock.disconnectFromServer()
            return True
        return False

    def listen(self, on_show: Callable[[], None]) -> bool:
        """Arranca el servidor de esta instancia (la primera).

        Devuelve True si quedó escuchando. Si `listen` falla (pipe huérfano),
        limpia con removeServer y reintenta UNA vez; si vuelve a fallar, False.
        """
        for _attempt in range(2):
            server = self._server_factory()
            server.removeServer(self._key)  # limpia un pipe huérfano si quedó
            if server.listen(self._key):
                self._server = server
                server.newConnection.connect(
                    lambda: self._on_connection(on_show))
                return True
            server.removeServer(self._key)
        return False

    def _on_connection(self, on_show: Callable[[], None]) -> None:
        conn = self._server.nextPendingConnection()
        if conn is None:
            return
        on_show()
        conn.disconnectFromServer()
        conn.deleteLater()

    def close(self) -> None:
        if self._server is not None:
            self._server.close()
            self._server = None
```

```python
    def __init__(self, api_key: str = "", clock=time.monotonic) -> None:
        self._api_key = api_key
        self._clock = clock
        self._cache: dict[str, object] = {}   # path -> (m,total) | _UNKNOWN | _ERROR
        self._queued: set[str] = set()
        self._retry_at: dict[str, float] = {}  # path -> instante (clock) del próx. intento
        self._fails: dict[str, int] = {}
        self._lock = threading.Lock()
        self._q: "queue.Queue[str]" = queue.Queue()
        self._worker: threading.Thread | None = None

    def set_api_key(self, key: str) -> None:
        """Cambia la key. Los errores (cacheados o en backoff) se descartan: con
        otra key pueden andar. Los resultados válidos se conservan."""
        with self._lock:
            self._api_key = key
            for p in [p for p, v in self._cache.items() if v == _ERROR]:
                del self._cache[p]
            self._retry_at.clear()
            self._fails.clear()

    def lookup(self, exe_path: str) -> tuple[int, int] | None:
        if not exe_path or not self._api_key:
            return None
        with self._lock:
            cached = self._cache.get(exe_path)
            if isinstance(cached, tuple):
                return cached
            if cached is not None:        # _UNKNOWN o _ERROR: no reintentar
                return None
            if exe_path in self._queued:
                return None
            if self._clock() < self._retry_at.get(exe_path, 0.0):
                return None               # en backoff por un error transitorio
            self._queued.add(exe_path)
            # put + asegurar worker BAJO EL MISMO LOCK con el que el worker decide
            # salir por idle: nunca queda un item en la cola sin worker vivo.
            self._q.put(exe_path)
            self._ensure_worker_locked()
        return None

    def _ensure_worker_locked(self) -> None:
        """Arranca el worker si no hay uno vivo. Llamar con self._lock tomado."""
        if self._worker is not None and self._worker.is_alive():
            return
        self._worker = threading.Thread(target=self._run, daemon=True)
        self._worker.start()

    def _run(self) -> None:
        while True:
            try:
                path = self._q.get(timeout=_IDLE_TIMEOUT)
            except queue.Empty:
                with self._lock:
                    if self._q.empty():   # nadie encoló mientras decidíamos salir
                        self._worker = None
                        return
                continue
            self._process(path)
            time.sleep(_MIN_INTERVAL)  # respeta el rate limit del free tier

    def _process(self, path: str) -> None:
        """Consulta un path y actualiza cache/backoff. Los errores de red o HTTP
        (401/403/429/5xx) son transitorios: NO se cachean, se reintenta luego."""
        failed = False
        result: object = None
        try:
            result = self._query(path)
        except Exception as exc:  # noqa: BLE001
            log.debug("VT fallo en %s: %s", path, exc)
            failed = True
        with self._lock:
            self._queued.discard(path)
            if failed:
                n = self._fails.get(path, 0) + 1
                self._fails[path] = n
                delay = min(_BACKOFF_BASE * (2 ** (n - 1)), _BACKOFF_MAX)
                self._retry_at[path] = self._clock() + delay
            else:
                self._fails.pop(path, None)
                self._retry_at.pop(path, None)
                self._cache[path] = result
```

```python
"""Utilidades de salida segura compartidas por los generadores de informes."""
from __future__ import annotations

# Un valor que empieza así lo interpretan Excel/LibreOffice como fórmula
# (CSV/formula injection). Los \t y \r también se usan para evadir el filtro.
_FORMULA_PREFIXES = ("=", "+", "-", "@", "\t", "\r")


def escape_formula(value):
    """Prefija con ' los textos que una planilla ejecutaría como fórmula."""
    if isinstance(value, str) and value.startswith(_FORMULA_PREFIXES):
        return "'" + value
    return value


def safe_filename(hint: str | None, default: str) -> str:
    """Solo el nombre de archivo del hint (sin rutas ni `..`); si queda vacío, `default`.

    Evita path traversal (`../../x`) y rutas absolutas. Soporta `/` y `\\`
    sin importar la plataforma, y neutraliza ':' (streams alternativos NTFS).
    """
    if not hint:
        return default
    name = hint.replace("\\", "/").rsplit("/", 1)[-1].strip().replace(":", "_")
    return default if name in ("", ".", "..") else name
```

```python
"""Rotación de informes: evita que `reports/` crezca sin límite."""
from __future__ import annotations

import logging
from pathlib import Path

log = logging.getLogger(__name__)

_PATTERNS = ("*_report_*.csv", "*_report_*.xlsx", "*_report_*.pdf")


def rotate_reports(directory: str | Path, keep: int = 20) -> list[Path]:
    """Borra los informes más viejos y deja los `keep` más recientes (por mtime)."""
    d = Path(directory)
    if not d.is_dir():
        return []
    files: list[Path] = []
    for pattern in _PATTERNS:
        files.extend(p for p in d.glob(pattern) if p.is_file())
    files.sort(key=lambda p: p.stat().st_mtime, reverse=True)
    removed: list[Path] = []
    for p in files[max(keep, 0):]:
        try:
            p.unlink()
            removed.append(p)
        except OSError as exc:
            log.debug("No se pudo borrar el informe viejo %s: %s", p, exc)
    return removed
```

```python
    def bind_tray(self, tray: QSystemTrayIcon) -> None:
        self._tray = tray
        tray.messageClicked.connect(self._on_clicked)

    def notify(self, title: str, message: str) -> None:
        self._signal.emit(title, message, "")

    def notify_link(self, title: str, message: str, url: str) -> None:
        """Como notify(), pero el clic en el toast abre `url` (solo https)."""
        self._signal.emit(title, message, url)

        # El toast más reciente manda: pisa (o limpia) el link del anterior.
        self._click_url = url if url.startswith("https://") else None
        if self._tray is not None:
            self._tray.showMessage(title, message,
                                   QSystemTrayIcon.MessageIcon.Information, 6000)
```

```python
"""Geometría de ventana: que una posición guardada nunca deje la ventana inalcanzable.

Si el usuario desconecta un monitor (o cambia la resolución), la geometría
guardada puede quedar fuera de todas las pantallas. Lógica pura, sin Qt.
"""
from __future__ import annotations

from typing import Sequence

Rect = tuple[int, int, int, int]  # x, y, ancho, alto

_MIN_VISIBLE_W = 120   # ancho horizontal mínimo de la barra de título visible
_MIN_BELOW_TOP = 40    # el borde superior debe dejar al menos esto dentro de la pantalla


def _title_bar_reachable(rect: Rect, screen: Rect) -> bool:
    x, y, w, _h = rect
    sx, sy, sw, sh = screen
    overlap_w = min(x + w, sx + sw) - max(x, sx)
    return overlap_w >= min(_MIN_VISIBLE_W, w) and sy <= y <= sy + sh - _MIN_BELOW_TOP


def clamp_to_screens(rect: Rect, screens: Sequence[Rect]) -> Rect:
    """`rect` si su barra de título es alcanzable en alguna pantalla; si no, lo
    centra en `screens[0]` (la primaria), achicándolo si no entra."""
    if not screens:
        return rect
    if any(_title_bar_reachable(rect, s) for s in screens):
        return rect
    sx, sy, sw, sh = screens[0]
    w = min(rect[2], sw)
    h = min(rect[3], sh)
    return (sx + (sw - w) // 2, sy + (sh - h) // 2, w, h)


def available_screen_rects() -> list[Rect]:
    """`availableGeometry()` de cada pantalla (excluye taskbar), primaria primero."""
    from PySide6.QtGui import QGuiApplication

    primary = QGuiApplication.primaryScreen()
    screens = sorted(QGuiApplication.screens(), key=lambda s: s is not primary)
    out: list[Rect] = []
    for s in screens:
        g = s.availableGeometry()
        out.append((g.x(), g.y(), g.width(), g.height()))
    return out
```
